def load_and_preprocess(df):
    """
    Advanced preprocessing pipeline focusing on data integrity.
    Includes handling zero prices, categorical imputation, color standardization,
    numerical outlier treatment (winsorization), and temporal feature engineering.
    """
    # Create a copy to prevent SettingWithCopyWarning
    data = df.copy()

    # A. Handling Zero Prices (Price on Call)
    # Strategy: Impute 0 prices using the median price of that specific car (Title + Model Year)
    # If no match, use the median of that specific Model Year.
    data['price'] = data['price'].replace(0, np.nan)
    data['price'] = data.groupby(['title', 'model'])['price'].transform(lambda x: x.fillna(x.median()))
    data['price'] = data['price'].fillna(data.groupby('model')['price'].transform('median'))

    # Ensure remaining NaNs in price are handled, e.g., by global median or dropping
    # For robustness, fill any remaining price NaNs with global median if groups fail
    data['price'].fillna(data['price'].median(), inplace=True)

    # B. Professional Imputation for "Not Available" Categories
    # Most cars in Pakistan are Petrol/Manual/Local. We use grouped modes for accuracy.
    categorical_cols = ['transmission', 'fuel_type', 'assembly']
    for col in categorical_cols:
        data[col] = data[col].replace('Not Available', np.nan)
        # Impute based on title (e.g., "Suzuki Alto" should usually be "Manual" if unspecified)
        data[col] = data.groupby('title')[col].transform(lambda x: x.fillna(x.mode()[0] if not x.mode().empty else 'Unknown'))
        # Fallback for any remaining NaNs after grouped imputation (e.g., if a title has all NaNs)
        data[col].fillna(data[col].mode()[0] if not data[col].mode().empty else 'Unknown', inplace=True)

    # C. Standardizing Colors (Handling the 809 unique color names)
    # Simplify top colors and group rare ones as 'Other'
    top_colors = data['color'].value_counts().nlargest(10).index
    data['color_group'] = data['color'].apply(lambda x: x if x in top_colors else 'Other')

    # D. Numerical Outlier Treatment (Winsorization) and Cleaned Columns
    # Apply winsorization to price, mileage, and engine_capacity to handle extreme outliers
    print("Applying winsorization for numerical features...")
    data['price_clean'] = winsorize(data['price'], limits=[0.01, 0.01]) # Cap at 1st and 99th percentile
    data['mileage_clean'] = winsorize(data['mileage'], limits=[0.01, 0.01])
    data['engine_capacity_clean'] = winsorize(data['engine_capacity'], limits=[0.01, 0.01])
    print("Numerical features winsorized and '_clean' columns created.")

    # E. Temporal Feature Engineering: Convert post_date to datetime and calculate days_since_post
    print("Converting 'post_date' to datetime and calculating 'days_since_post'...")
    data['post_date'] = pd.to_datetime(data['post_date'])
    # Assuming 'today' is the most recent date in the dataset for a consistent 'post_date' reference
    latest_date_in_data = data['post_date'].max()
    data['days_since_post'] = (latest_date_in_data - data['post_date']).dt.days
    print("'post_date' converted and 'days_since_post' created.")

    print("\n--- Preprocessing Summary ---")
    print("Missing values after imputation:")
    print(data[['price', 'transmission', 'fuel_type', 'assembly']].isnull().sum())
    print("\nFirst 5 rows of cleaned data with new features:")
    display(data[['price', 'price_clean', 'mileage', 'mileage_clean', 'engine_capacity', 'engine_capacity_clean', 'fuel_type', 'transmission', 'assembly', 'color_group', 'post_date', 'days_since_post']].head())
    print("\nDescriptive statistics for cleaned numerical columns:")
    display(data[['price_clean', 'mileage_clean', 'engine_capacity_clean']].describe())

    return data

df_clean = load_and_preprocess(df)

def engineer_features(data):
    """
    Create sophisticated features for market analysis
    """
    print("="*80)
    print("🛠️  PHASE 2: ADVANCED FEATURE ENGINEERING")
    print("="*80 + "\n")

    df = data.copy()

    # A. Brand Extraction (Make)
    print("Creating 'make' feature...")
    df['make'] = df['title'].apply(lambda x: x.split(' ')[0])

    # B. Engine Segment (CC Tiers)
    print("Creating 'market_segment' feature...")
    bins = [0, 700, 1100, 1600, 2000, 20000]
    labels = ['Ultra-Economy', 'Hatchback', 'Sedan/Crossover', 'Executive', 'Luxury/Heavy']
    df['market_segment'] = pd.cut(df['engine_capacity_clean'], bins=bins, labels=labels)

    # C. Regional Hub Classification
    print("Classifying Regional Hubs...")
    major_hubs = ['Lahore', 'Karachi', 'Islamabad', 'Rawalpindi', 'Peshawar']
    df['city_tier'] = df['city'].apply(
        lambda x: 'Tier-1' if x in major_hubs else 'Tier-2/3'
    )

    # -------------------------------------------------------------------------
    # 2.1 Value Retention Index (VRI)
    # -------------------------------------------------------------------------
    print("2.1 Engineering Value Retention Index (VRI)...")

    # VRI = Price / (log(Mileage+1) * (Age+1))
    # Higher VRI = Better value retention
    df['vri'] = df['price_clean'] / (np.log1p(df['mileage_clean']) * (df['vehicle_age'] + 1))

    print(f"  ✓ VRI Stats: Mean={df['vri'].mean():.2f}, Median={df['vri'].median():.2f}")

    # -------------------------------------------------------------------------
    # 2.2 Usage Intensity
    # -------------------------------------------------------------------------
    print("\n2.2 Calculating Usage Intensity...")

    df['usage_intensity'] = df['mileage_clean'] / (df['vehicle_age'] + 1)

    # Categorize usage
    df['usage_category'] = pd.cut(
        df['usage_intensity'],
        bins=[0, 10000, 20000, np.inf],
        labels=['Light', 'Moderate', 'Heavy']
    )

    print(f"  ✓ Usage Distribution:")
    print(df['usage_category'].value_counts().to_string())

    # -------------------------------------------------------------------------
    # 2.3 Price Per CC (Engine Efficiency Proxy)
    # -------------------------------------------------------------------------
    print("\n2.3 Computing Price-to-Engine Ratio...")

    df['price_per_cc'] = df['price_clean'] / df['engine_capacity_clean']

    # -------------------------------------------------------------------------
    # 2.5 Depreciation Rate
    # -------------------------------------------------------------------------
    print("\n2.5 Modeling Depreciation Rate...")

    # Estimate original price using newest similar vehicles (age 0 or 1, and in the same segment)
    # Fallback to general segment quantile if no new cars for make/segment
    new_car_prices = df[df['vehicle_age'] <= 1].groupby(['make', 'market_segment'])['price_clean'].transform(lambda x: x.quantile(0.90))
    df['estimated_original_price'] = new_car_prices.fillna(df.groupby('market_segment')['price_clean'].transform(lambda x: x.quantile(0.90)))

    df['depreciation_pct'] = ((df['estimated_original_price'] - df['price_clean']) /
                               df['estimated_original_price'] * 100).clip(0, 100)

    df['annual_depreciation'] = df['depreciation_pct'] / (df['vehicle_age'] + 1)

    print(f"  ✓ Avg Annual Depreciation: {df['annual_depreciation'].median():.2f}%")

    # -------------------------------------------------------------------------
    # 2.6 Total Cost of Ownership (TCO)
    # -------------------------------------------------------------------------
    print("\n2.6 Calculating Total Cost of Ownership (5-Year)...")

    def calculate_tco(row):
        # Fuel efficiency estimates
        fuel_efficiency = {
            'Hybrid': 22.0,    # km/L
            'Petrol': 14.0,
            'Diesel': 16.0,
            'Unknown': 14.0 # Fallback for 'Unknown' fuel type
        }

        fuel_price = {
            'Hybrid': PETROL_PRICE_PKR,
            'Petrol': PETROL_PRICE_PKR,
            'Diesel': DIESEL_PRICE_PKR,
            'Unknown': PETROL_PRICE_PKR # Fallback for 'Unknown' fuel type
        }

        kmpl = fuel_efficiency.get(row['fuel_type'], 14.0)
        price_per_liter = fuel_price.get(row['fuel_type'], PETROL_PRICE_PKR)

        # Annual costs
        annual_fuel = (ANNUAL_KM_AVG / kmpl) * price_per_liter
        annual_maintenance = row['price_clean'] * 0.02  # 2% of vehicle value

        # 5-year TCO
        tco = row['price_clean'] + (annual_fuel * OWNERSHIP_YEARS) + (annual_maintenance * OWNERSHIP_YEARS)

        return tco

    df['tco_5yr'] = df.apply(calculate_tco, axis=1)

    print(f"  ✓ TCO Range: {df['tco_5yr'].min()/1e6:.2f}M - {df['tco_5yr'].max()/1e6:.2f}M PKR")

    print("\n✅ FEATURE ENGINEERING COMPLETE\n")

    return df

def perform_eda(df):
    """
    Advanced statistical exploration with hypothesis testing
    """
    print("="*80)
    print("📊 PHASE 3: ADVANCED EXPLORATORY DATA ANALYSIS")
    print("="*80 + "\n")

    # -------------------------------------------------------------------------
    # 3.1 Distribution Analysis
    # -------------------------------------------------------------------------
    print("3.1 KEY DISTRIBUTION METRICS")
    print("-" * 80)

    summary_stats = df[['price_clean', 'mileage_clean', 'vehicle_age', 'vri']].describe()
    print(summary_stats.to_string())

    # -------------------------------------------------------------------------
    # 3.2 Statistical Hypothesis Testing
    # -------------------------------------------------------------------------
    print("\n\n3.2 HYPOTHESIS TESTING RESULTS")
    print("-" * 80)

    # Test 1: Price difference between Local vs Imported (t-test)
    local = df[df['assembly'] == 'Local']['price_clean']
    imported = df[df['assembly'] == 'Imported']['price_clean']

    t_stat, p_value = stats.ttest_ind(local, imported)
    print(f"\n📈 Test 1: Price ~ Assembly Type")
    print(f"  H0: No price difference between Local and Imported")
    print(f"  t-statistic: {t_stat:.4f}, p-value: {p_value:.4e}")
    print(f"  Result: {'REJECT H0' if p_value < 0.05 else 'FAIL TO REJECT H0'} at \u03b1=0.05")
    print(f"  Local Median: Rs. {local.median()/1e6:.2f}M | Imported Median: Rs. {imported.median()/1e6:.2f}M")

    # Test 2: ANOVA for price across fuel types
    petrol = df[df['fuel_type'] == 'Petrol']['price_clean']
    hybrid = df[df['fuel_type'] == 'Hybrid']['price_clean']
    diesel = df[df['fuel_type'] == 'Diesel']['price_clean']

    # Handle cases where a fuel type might be empty after filtering
    fuel_types_for_anova = []
    if not petrol.empty: fuel_types_for_anova.append(petrol)
    if not hybrid.empty: fuel_types_for_anova.append(hybrid)
    if not diesel.empty: fuel_types_for_anova.append(diesel)

    if len(fuel_types_for_anova) >= 2: # ANOVA requires at least two groups
        f_stat, p_value_anova = f_oneway(*fuel_types_for_anova)
        print(f"\n📈 Test 2: Price ~ Fuel Type (ANOVA)")
        print(f"  H0: No price difference across fuel types")
        print(f"  F-statistic: {f_stat:.4f}, p-value: {p_value_anova:.4e}")
        print(f"  Result: {'REJECT H0' if p_value_anova < 0.05 else 'FAIL TO REJECT H0'} at \u03b1=0.05")
    else:
        print("\n📈 Test 2: Price ~ Fuel Type (ANOVA) - Not enough fuel types for ANOVA after filtering.")

    # Test 3: Correlation between Age and Price
    corr_coef, p_value_corr = stats.pearsonr(df['vehicle_age'], df['price_clean'])
    print(f"\n📈 Test 3: Correlation (Age vs Price)")
    print(f"  Pearson's r: {corr_coef:.4f}, p-value: {p_value_corr:.4e}")
    print(f"  Interpretation: {'Strong' if abs(corr_coef) > 0.7 else 'Moderate' if abs(corr_coef) > 0.4 else 'Weak'} {'negative' if corr_coef < 0 else 'positive'} correlation")

    # -------------------------------------------------------------------------
    # 3.3 Market Concentration Analysis
    # -------------------------------------------------------------------------
    print("\n\n3.3 MARKET CONCENTRATION METRICS")
    print("-" * 80)

    # Top 10 brands by count
    top_brands = df['make'].value_counts().head(10)
    total_cars = len(df)

    print("\n🚗 Top 10 Brands by Market Share:")
    for rank, (brand, count) in enumerate(top_brands.items(), 1):
        share = count / total_cars * 100
        print(f"  {rank:2d}. {brand:15s}: {count:6,} units ({share:5.2f}%)")

    # Herfindahl-Hirschman Index (HHI) for market concentration
    market_shares = (df['make'].value_counts() / total_cars) ** 2
    hhi = market_shares.sum() * 10000

    print(f"\n📊 Herfindahl-Hirschman Index (HHI): {hhi:.2f}")
    print(f"  Interpretation: {'Highly Concentrated' if hhi > 2500 else 'Moderately Concentrated' if hhi > 1500 else 'Competitive'} market")

    print("\n✅ EDA COMPLETE\n")

    return df

df_final = perform_eda(df_final)

def generate_business_insights(df):
    """
    Extract actionable business intelligence for PakWheels
    """
    print("="*80)
    print("💡 PHASE 5: BUSINESS INTELLIGENCE GENERATION")
    print("="*80 + "\n")

    insights = {}

    # -------------------------------------------------------------------------
    # 5.1 Price Elasticity Analysis (Renamed to Brand Depreciation Rate Analysis)
    # -------------------------------------------------------------------------
    print("5.1 BRAND DEPRECIATION RATE ANALYSIS")
    print("-" * 80)

    top_brands = df['make'].value_counts().head(5).index

    for brand in top_brands:
        brand_data = df[df['make'] == brand]

        # Use annual_depreciation directly for robustness
        median_annual_dep = brand_data['annual_depreciation'].median()
        print(f"  {brand:15s}: Median Annual Depreciation = {median_annual_dep:.2f}%")

    # -------------------------------------------------------------------------
    # 5.2 Market Share Trends
    # -------------------------------------------------------------------------
    print("\n\n5.2 MARKET DYNAMICS")
    print("-" * 80)

    # Recent vs older listings (days_since_post is now available)
    recent = df[df['days_since_post'] <= 15]
    older = df[df['days_since_post'] > 15]

    print(f"\n📊 Recent Listings (Last 15 days): {len(recent):,}")
    print(f"📊 Older Listings: {len(older):,}")

    print("\n🔥 Trending Brands (Recent vs Historical):")
    recent_top5 = recent['make'].value_counts().head(5)
    overall_top5 = df['make'].value_counts().head(5)

    for brand in recent_top5.index:
        recent_share = (recent_top5[brand] / len(recent)) * 100
        overall_share = (overall_top5.get(brand, 0) / len(df)) * 100
        trend = "📈" if recent_share > overall_share else "📉"
        print(f"  {trend} {brand:15s}: Recent {recent_share:.2f}% | Overall {overall_share:.2f}%")

    # -------------------------------------------------------------------------
    # 5.3 Regional Opportunity Analysis
    # -------------------------------------------------------------------------
    print("\n\n5.3 REGIONAL MARKET OPPORTUNITIES")
    print("-" * 80)

    tier1_avg = df[df['city_tier'] == 'Tier-1']['price_clean'].median()
    tier2_avg = df[df['city_tier'] == 'Tier-2/3']['price_clean'].median()

    premium_pct = ((tier1_avg - tier2_avg) / tier2_avg) * 100

    print(f"\n💰 Tier-1 Cities Median Price: Rs. {tier1_avg/1e6:.2f}M")
    print(f"💰 Tier-2/3 Cities Median Price: Rs. {tier2_avg/1e6:.2f}M")
    print(f"📊 Premium Differential: {premium_pct:+.2f}%")

    print(f"\n🎯 Strategic Insight:")
    print(f"   Vehicles in Tier-1 cities command a {premium_pct:.1f}% premium.")
    print(f"   Opportunity: Arbitrage between cities for price-sensitive buyers.")

    # -------------------------------------------------------------------------
    # 5.4 Fuel Type Transition Analysis
    # -------------------------------------------------------------------------
    print("\n\n5.4 GREEN TECHNOLOGY ADOPTION")
    print("-" * 80)

    fuel_trend = df.groupby(['model', 'fuel_type']).size().unstack(fill_value=0)

    if 'Hybrid' in fuel_trend.columns:
        recent_years = fuel_trend.loc[fuel_trend.index >= 2020]
        # Handle potential division by zero if no pre-2020 hybrids
        pre_2020_hybrids = fuel_trend.loc[fuel_trend.index < 2020, 'Hybrid'].sum()
        if pre_2020_hybrids > 0:
            hybrid_growth = recent_years['Hybrid'].sum() / pre_2020_hybrids
        else:
            hybrid_growth = float('inf') if recent_years['Hybrid'].sum() > 0 else 0

        print(f"\n🌱 Hybrid Market Analysis:")
        print(f"   Pre-2020 Hybrid Listings: {pre_2020_hybrids:,}")
        print(f"   2020+ Hybrid Listings: {recent_years['Hybrid'].sum():,}")

        if hybrid_growth == float('inf'):
            print(f"   Growth Factor: Infinite (No pre-2020 Hybrid listings found)")
        elif hybrid_growth > 0:
            print(f"   Growth Factor: {hybrid_growth:.2f}x")

        if hybrid_growth > 1:
            print(f"\n🎯 Insight: Hybrid adoption accelerating post-2020")

    # -------------------------------------------------------------------------
    # 5.5 Value Retention Leaders
    # -------------------------------------------------------------------------
    print("\n\n5.5 VALUE RETENTION CHAMPIONS")
    print("-" * 80)

    vri_by_brand = df.groupby('make')['vri'].median().sort_values(ascending=False).head(10)

    print(f"\n🏆 Top 10 Brands by Value Retention Index:")
    for rank, (brand, vri) in enumerate(vri_by_brand.items(), 1):
        print(f"  {rank:2d}. {brand:15s}: VRI = {vri:.2f}")

    print("\n✅ BUSINESS INTELLIGENCE COMPLETE\n")

    return insights

generate_business_insights(df_final)

# 1. DATA INGESTION & ROBUST CLEANING
# =================================================================
def load_and_preprocess(df):
    """
    Advanced preprocessing pipeline focusing on data integrity.
    """
    # Create a copy to prevent SettingWithCopyWarning
    data = df.copy()

    # A. Handling Zero Prices (Price on Call)
    # Strategy: Impute 0 prices using the median price of that specific car (Title + Model Year)
    # If no match, use the median of that specific Model Year.
    data['price'] = data['price'].replace(0, np.nan)
    data['price'] = data.groupby(['title', 'model'])['price'].transform(lambda x: x.fillna(x.median()))
    data['price'] = data['price'].fillna(data.groupby('model')['price'].transform('median'))

    # B. Professional Imputation for "Not Available" Categories
    # Most cars in Pakistan are Petrol/Manual/Local. We use grouped modes for accuracy.
    categorical_cols = ['transmission', 'fuel_type', 'assembly']
    for col in categorical_cols:
        data[col] = data[col].replace('Not Available', np.nan)
        # Impute based on title (e.g., "Suzuki Alto" should usually be "Manual" if unspecified)
        data[col] = data.groupby('title')[col].transform(lambda x: x.fillna(x.mode()[0] if not x.mode().empty else 'Unknown'))

    # C. Standardizing Colors (Handling the 809 unique color names)
    # Simplify top colors and group rare ones as 'Other'
    top_colors = data['color'].value_counts().nlargest(10).index
    data['color_group'] = data['color'].apply(lambda x: x if x in top_colors else 'Other')

    return data

# 2. ADVANCED FEATURE ENGINEERING (THE SENIOR EDGE)
# =================================================================
def engineer_features(data):
    """
    Creating Latent Variables to describe market behavior.
    """
    # A. Brand Extraction (Make)
    data['make'] = data['title'].apply(lambda x: x.split(' ')[0])

    # B. Engine Segment (CC Tiers)
    # 660cc (Economy), 1300cc (Family), 1800cc (Executive), 2000cc+ (Luxury/SUV)
    bins = [0, 700, 1100, 1600, 2000, 20000]
    labels = ['Ultra-Economy', 'Hatchback', 'Sedan/Crossover', 'Executive', 'Luxury/Heavy']
    data['market_segment'] = pd.cut(data['engine_capacity'], bins=bins, labels=labels)

    # C. Usage Intensity (Kilometers driven per year)
    # High intensity (>20k km/year) indicates commercial/heavy commute
    data['usage_intensity'] = data['mileage'] / (data['vehicle_age'] + 1)

    # D. Value Retention Index (VRI)
    # Formula: Price / (Log(Mileage) * Age). High VRI = Low Depreciation.
    data['vri'] = data['price'] / (np.log1p(data['mileage']) * (data['vehicle_age'] + 1))

    # E. Regional Hub Clustering
    major_hubs = ['Lahore', 'Karachi', 'Islamabad', 'Rawalpindi', 'Peshawar']
    data['is_major_hub'] = data['city'].apply(lambda x: 'Major Hub' if x in major_hubs else 'Secondary City')

    return data

# This addresses the EV/Hybrid research requirement in the Job Description.
def calculate_tco(row):
    # Estimated Fuel Avg (KM/L) based on data and market standards
    if row['fuel_type'] == 'Hybrid':
        fuel_avg = 22.0  # High efficiency
        maint_premium = 1.2 # 20% higher due to specialized parts/battery
    else:
        fuel_avg = 14.0  # Standard petrol
        maint_premium = 1.0

    annual_fuel_cost = (ANNUAL_KM_AVG / fuel_avg) * PETROL_PRICE_2026
    annual_maint = (row['price'] * 0.02) * maint_premium # 2% of car value per year

    # TCO over 5 years
    tco_5yr = row['price'] + (annual_fuel_cost * OWNERSHIP_HORIZON) + (annual_maint * OWNERSHIP_HORIZON)
    return tco_5yr
