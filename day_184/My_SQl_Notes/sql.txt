-- Step 1: Drop the existing stuID column
ALTER TABLE students DROP COLUMN stuID;

-- Step 2: Add a new stuID column as an auto-increment primary key
ALTER TABLE students ADD COLUMN stuID INT AUTO_INCREMENT PRIMARY KEY;

This approach ensures that the stuID column is reset to start from 1 and is automatically incremented for each new record added to the table. It also maintains referential integrity and avoids any potential issues associated with manually reassigning IDs.
